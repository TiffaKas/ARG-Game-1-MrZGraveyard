function saveData() { const nameInput = document.getElementById('lastName').value.trim(); const bio = document.getElementById('bioData').value.trim(); const desc = document.getElementById('desc').value.trim(); // 获取生平 const task = localStorage.getItem('current_task_content'); if (!task) { feedback("别那么无聊"); return; } if (!nameInput || !bio) { feedback("好像不太对"); return; } const pureName = nameInput.split('（')[0].split('(')[0].trim(); const keys = MASTER_KEYS[pureName]; if(!keys) { feedback("好像不太对"); return; } const matchCount = keys.filter(k => bio.includes(k)).length; if(matchCount < 2) { feedback("好像不太对"); return; } let archives = JSON.parse(localStorage.getItem('cemetery_files') || '[]'); // 重点：这里增加了 bio 和 desc 的存储 archives.push({ name: nameInput, time: new Date().toLocaleString('zh-CN', { hour12: false }), bio: bio, desc: desc }); localStorage.setItem('cemetery_files', JSON.stringify(archives)); localStorage.removeItem('current_task_content'); localStorage.setItem('last_record_timestamp', Date.now()); window.location.href = 'system.html'; }
